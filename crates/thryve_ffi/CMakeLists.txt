# CMakeLists.txt for the Thryve FFI example applications
#
# Usage (from crates/thryve_ffi, after `cargo build -p thryve_ffi`):
#   cmake -S . -B build
#   cmake --build build
#
# For a release build of the library, build it with `cargo build --release -p thryve_ffi` and
# configure with -DCMAKE_BUILD_TYPE=Release (single-config generators such as Ninja or Make).
#
# What the cargo build produces, in the workspace's target/debug/ or target/release/:
#   Windows, MSVC toolchain:  thryve_ffi.dll and its import library, thryve_ffi.dll.lib
#   Windows, GNU toolchain:   thryve_ffi.dll and its import library, libthryve_ffi.dll.a
#   Linux / macOS:            libthryve_ffi.so / libthryve_ffi.dylib
# (thryve_ffi.lib and libthryve_ffi.a are static libraries, which also need Rust's system
# libraries to link; these examples use the shared library.)
#
# The headers (thryve.h and thryve_ffi.h) are generated by build.rs and copied to
# crates/thryve_ffi/out/.
#
# Each path can be set with -D on the cmake command line:
#   THRYVE_FFI_TARGET_DIR  — Cargo's target folder (default: the workspace's target/)
#   THRYVE_FFI_LIB_DIR     — The folder with the library files (default: <target>/debug or /release)
#   THRYVE_FFI_INCLUDE_DIR — The folder with thryve.h and thryve_ffi.h (default: out/)

cmake_minimum_required(VERSION 3.15)
project(thryve_ffi_example LANGUAGES C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)

# ============================================================
# Find the Thryve FFI library and headers
# ============================================================

set(THRYVE_FFI_TARGET_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../target"
    CACHE PATH "Cargo's target folder")

if(CMAKE_BUILD_TYPE STREQUAL "Release")
    set(THRYVE_FFI_PROFILE "release")
else()
    set(THRYVE_FFI_PROFILE "debug")
endif()

set(THRYVE_FFI_LIB_DIR "${THRYVE_FFI_TARGET_DIR}/${THRYVE_FFI_PROFILE}"
    CACHE PATH "The folder with the Thryve FFI library files")
set(THRYVE_FFI_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/out"
    CACHE PATH "The folder with thryve.h and thryve_ffi.h")

if(NOT EXISTS "${THRYVE_FFI_INCLUDE_DIR}/thryve.h")
    message(FATAL_ERROR "thryve.h not found in ${THRYVE_FFI_INCLUDE_DIR}. "
                        "Run 'cargo build -p thryve_ffi' first.")
endif()

# ============================================================
# Define the Thryve FFI target: the shared library
# ============================================================

if(WIN32)
    set(THRYVE_FFI_DLL "${THRYVE_FFI_LIB_DIR}/thryve_ffi.dll")

    # The DLL's import library: the MSVC toolchain names it thryve_ffi.dll.lib, the GNU
    # toolchain libthryve_ffi.dll.a.
    if(EXISTS "${THRYVE_FFI_LIB_DIR}/thryve_ffi.dll.lib")
        set(THRYVE_FFI_IMPORT_LIB "${THRYVE_FFI_LIB_DIR}/thryve_ffi.dll.lib")
    else()
        set(THRYVE_FFI_IMPORT_LIB "${THRYVE_FFI_LIB_DIR}/libthryve_ffi.dll.a")
    endif()

    if(NOT EXISTS "${THRYVE_FFI_DLL}" OR NOT EXISTS "${THRYVE_FFI_IMPORT_LIB}")
        message(FATAL_ERROR "thryve_ffi.dll or its import library not found in "
                            "${THRYVE_FFI_LIB_DIR}. Run 'cargo build -p thryve_ffi' first.")
    endif()

    add_library(thryve_ffi SHARED IMPORTED)
    set_target_properties(thryve_ffi PROPERTIES
        IMPORTED_LOCATION "${THRYVE_FFI_DLL}"
        IMPORTED_IMPLIB "${THRYVE_FFI_IMPORT_LIB}"
    )
elseif(APPLE)
    add_library(thryve_ffi SHARED IMPORTED)
    set_target_properties(thryve_ffi PROPERTIES
        IMPORTED_LOCATION "${THRYVE_FFI_LIB_DIR}/libthryve_ffi.dylib"
    )
else()
    add_library(thryve_ffi SHARED IMPORTED)
    set_target_properties(thryve_ffi PROPERTIES
        IMPORTED_LOCATION "${THRYVE_FFI_LIB_DIR}/libthryve_ffi.so"
    )
endif()

# Whatever links thryve_ffi also gets its headers.
set_target_properties(thryve_ffi PROPERTIES
    INTERFACE_INCLUDE_DIRECTORIES "${THRYVE_FFI_INCLUDE_DIR}"
)

# ============================================================
# The examples: main.c includes thryve.h, main.cpp the extern "C" wrapper thryve_ffi.h
# ============================================================

add_executable(thryve_ffi_example_c examples/main.c)
target_link_libraries(thryve_ffi_example_c PRIVATE thryve_ffi)

add_executable(thryve_ffi_example_cpp examples/main.cpp)
target_link_libraries(thryve_ffi_example_cpp PRIVATE thryve_ffi)

# On Windows, an executable finds the DLL next to it.
if(WIN32)
    foreach(example thryve_ffi_example_c thryve_ffi_example_cpp)
        add_custom_command(TARGET ${example} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different
                "${THRYVE_FFI_DLL}"
                $<TARGET_FILE_DIR:${example}>
            COMMENT "Copying thryve_ffi.dll next to ${example}"
        )
    endforeach()
endif()

# ============================================================
# Installation targets
# ============================================================

install(TARGETS thryve_ffi_example_c thryve_ffi_example_cpp
    RUNTIME DESTINATION bin
)

# On Windows, the DLL goes next to the executables, and its import library to lib/.
if(WIN32)
    install(FILES "${THRYVE_FFI_DLL}" DESTINATION bin)
    install(FILES "${THRYVE_FFI_IMPORT_LIB}" DESTINATION lib)
endif()

# Install the headers
install(FILES "${THRYVE_FFI_INCLUDE_DIR}/thryve.h"
        "${THRYVE_FFI_INCLUDE_DIR}/thryve_ffi.h"
    DESTINATION include
)
